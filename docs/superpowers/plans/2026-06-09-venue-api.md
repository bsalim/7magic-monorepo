def require_website_read_access(
    request: Request,
    settings: Annotated[Settings, Depends(get_settings)],
) -> None:
    if settings.venue_read_allowed_origins:
        origin = request.headers.get("origin")
        referer = request.headers.get("referer")
        referer_allowed = any(
            referer and referer.startswith(allowed)
            for allowed in settings.venue_read_allowed_origins
        )
        if origin not in settings.venue_read_allowed_origins and not referer_allowed:
            raise AuthError(
                status_code=status.HTTP_403_FORBIDDEN,
                code="website_access_denied",
                message="Website venue access is not allowed from this origin.",
            )

    if settings.venue_read_api_key:
        provided_key = request.headers.get("x-7magic-venue-key")
        if provided_key != settings.venue_read_api_key:
            raise AuthError(
                status_code=status.HTTP_403_FORBIDDEN,
                code="website_access_denied",
                message="Website venue access key is invalid.",
            )

from __future__ import annotations

from decimal import Decimal
from math import ceil
from typing import Any
from uuid import UUID

from sqlalchemy import func, select
from sqlalchemy.exc import IntegrityError
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy.orm import selectinload

from app.models import Venue, VenuePhoto
from app.schemas.content import (
    ImageRef,
    Pagination,
    VenueAdminSummary,
    VenueCreate,
    VenueDetail,
    VenueListResponse,
    VenuePhotoResponse,
    VenueUpdate,
)


class VenueSlugConflictError(ValueError):
    pass


class VenueNotFoundError(LookupError):
    pass


def decimal_to_int(value: Decimal | int | None) -> int | None:
    if value is None:
        return None
    return int(value)


def photo_to_gallery_item(photo: VenuePhoto) -> dict[str, Any]:
    url = photo.cdn_url or ""
    thumbnail_url = photo.thumbnail_url or url
    return {
        "id": photo.id,
        "url": url,
        "thumbnail_url": thumbnail_url,
        "alt_text": photo.alt_text,
        "sort_order": photo.sort_order,
        "webp": url,
        "fallback": url,
        "thumbWebp": thumbnail_url,
        "thumbFallback": thumbnail_url,
    }


def cover_photo_for(venue: Venue) -> ImageRef:
    first_photo = venue.photos[0] if venue.photos else None
    if first_photo and first_photo.cdn_url:
        return ImageRef(
            alt=first_photo.alt_text or f"{venue.name} wedding venue",
            small_url=first_photo.thumbnail_url or first_photo.cdn_url,
            large_url=first_photo.cdn_url,
        )
    return ImageRef(
        alt=f"{venue.name} wedding venue",
        small_url="/img/wedding-venue-deal-768.jpg",
        large_url="/img/wedding-venue-deal-1920.webp",
    )


def venue_detail_from_model(venue: Venue) -> VenueDetail:
    return VenueDetail(
        id=venue.id,
        name=venue.name,
        slug=venue.slug,
        city=venue.city,
        district=venue.district,
        stars=venue.stars,
        price_start_from=decimal_to_int(venue.price_start_from),
        price_for_total_pax=venue.price_for_total_pax,
        path_url=venue.path_url,
        cover_photo=cover_photo_for(venue),
        address=venue.address,
        description=venue.description or "",
        status=venue.status,
        gallery=[photo_to_gallery_item(photo) for photo in venue.photos],
        packages=venue.packages or [],
        seo={
            "title": f"{venue.name} - {venue.district}, {venue.city} | Wedding Venue",
            "meta_description": venue.description or "",
            "canonical_url": venue.path_url,
        },
    )


class VenueService:
    async def admin_list(self, session: AsyncSession) -> list[VenueAdminSummary]:
        result = await session.execute(
            select(Venue).options(selectinload(Venue.photos)).order_by(Venue.id.desc())
        )
        return [
            VenueAdminSummary(
                id=venue.id,
                name=venue.name,
                slug=venue.slug,
                city=venue.city,
                district=venue.district,
                stars=venue.stars,
                price_start_from=decimal_to_int(venue.price_start_from),
                price_for_total_pax=venue.price_for_total_pax,
                status=venue.status,
                cover_photo=cover_photo_for(venue),
            )
            for venue in result.scalars().unique()
        ]

    async def admin_detail(self, session: AsyncSession, venue_id: int) -> VenueDetail:
        venue = await self._get_venue_by_id(session, venue_id)
        return venue_detail_from_model(venue)

    async def create(self, session: AsyncSession, payload: VenueCreate) -> VenueDetail:
        venue = Venue(**payload.model_dump())
        session.add(venue)
        try:
            await session.commit()
        except IntegrityError as error:
            await session.rollback()
            raise VenueSlugConflictError("Venue slug already exists for this city.") from error
        return await self.admin_detail(session, venue.id)

    async def update(
        self,
        session: AsyncSession,
        venue_id: int,
        payload: VenueUpdate,
    ) -> VenueDetail:
        venue = await self._get_venue_by_id(session, venue_id)
        for key, value in payload.model_dump(exclude_unset=True).items():
            setattr(venue, key, value)
        try:
            await session.commit()
        except IntegrityError as error:
            await session.rollback()
            raise VenueSlugConflictError("Venue slug already exists for this city.") from error
        return await self.admin_detail(session, venue_id)

    async def archive(self, session: AsyncSession, venue_id: int) -> VenueDetail:
        venue = await self._get_venue_by_id(session, venue_id)
        venue.status = "archived"
        await session.commit()
        return await self.admin_detail(session, venue_id)

    async def website_list(
        self,
        session: AsyncSession,
        *,
        q: str | None,
        city: str | None,
        stars_min: int | None,
        page: int,
        page_size: int,
    ) -> VenueListResponse:
        filters = [Venue.status == "active"]
        if q:
            filters.append(func.lower(Venue.name).contains(q.casefold()))
        if city:
            filters.append(func.lower(Venue.city) == city.casefold())
        if stars_min:
            filters.append(Venue.stars >= stars_min)

        total = await session.scalar(select(func.count()).select_from(Venue).where(*filters)) or 0
        result = await session.execute(
            select(Venue)
            .where(*filters)
            .options(selectinload(Venue.photos))
            .order_by(Venue.id.desc())
            .offset((page - 1) * page_size)
            .limit(page_size)
        )
        items = [
            venue_detail_from_model(venue).model_dump(
                include={
                    "id",
                    "name",
                    "slug",
                    "city",
                    "district",
                    "stars",
                    "price_start_from",
                    "price_for_total_pax",
                    "path_url",
                    "cover_photo",
                }
            )
            for venue in result.scalars().unique()
        ]
        return VenueListResponse(
            items=items,
            pagination=Pagination(
                page=page,
                page_size=page_size,
                total=total,
                total_pages=ceil(total / page_size) if total else 0,
            ),
        )

    async def website_detail(self, session: AsyncSession, *, city: str, slug: str) -> VenueDetail:
        result = await session.execute(
            select(Venue)
            .where(
                func.lower(Venue.city) == city.casefold(),
                Venue.slug == slug,
                Venue.status == "active",
            )
            .options(selectinload(Venue.photos))
        )
        venue = result.scalars().unique().one_or_none()
        if venue is None:
            raise VenueNotFoundError("Venue not found.")
        return venue_detail_from_model(venue)

    async def attach_photo(
        self,
        session: AsyncSession,
        *,
        venue_id: int,
        storage_result: dict[str, Any],
        alt_text: str | None,
        sort_order: int,
    ) -> VenuePhotoResponse:
        await self._get_venue_by_id(session, venue_id)
        photo = VenuePhoto(
            venue_id=venue_id,
            sort_order=sort_order,
            alt_text=alt_text,
            filename=storage_result["filename"],
            original_filename=storage_result.get("original_filename"),
            content_type=storage_result.get("content_type"),
            original_file_size=storage_result.get("file_size"),
            storage_key=storage_result["storage_key"],
            cdn_url=storage_result["url"],
            thumbnail_url=storage_result["thumbnail_url"],
        )
        session.add(photo)
        await session.commit()
        await session.refresh(photo)
        return VenuePhotoResponse(
            id=photo.id,
            venue_id=photo.venue_id,
            temp_venue_id=str(photo.temp_venue_id) if isinstance(photo.temp_venue_id, UUID) else None,
            url=photo.cdn_url or "",
            thumbnail_url=photo.thumbnail_url or photo.cdn_url or "",
            alt_text=photo.alt_text,
            sort_order=photo.sort_order,
            filename=photo.filename or "",
            original_filename=photo.original_filename,
            content_type=photo.content_type,
            file_size=photo.original_file_size,
            storage_key=photo.storage_key or "",
            variants=storage_result.get("variants", {}),
        )

    async def _get_venue_by_id(self, session: AsyncSession, venue_id: int) -> Venue:
        result = await session.execute(
            select(Venue).where(Venue.id == venue_id).options(selectinload(Venue.photos))
        )
        venue = result.scalars().unique().one_or_none()
        if venue is None:
            raise VenueNotFoundError("Venue not found.")
        return venue


venue_service = VenueService()
```

from __future__ import annotations

from typing import Annotated

from fastapi import APIRouter, Depends, Query, status
from sqlalchemy.ext.asyncio import AsyncSession

from app.api.v1.dependencies import require_website_read_access
from app.core.database import get_db_session
from app.core.errors import error_response
from app.schemas.content import VenueDetail, VenueListResponse
from app.services.venues import VenueNotFoundError, venue_service

router = APIRouter(dependencies=[Depends(require_website_read_access)])


@router.get("", response_model=VenueListResponse)
async def list_venues(
    session: Annotated[AsyncSession, Depends(get_db_session)],
    q: Annotated[str | None, Query(min_length=1, max_length=100)] = None,
    city: Annotated[str | None, Query(min_length=1, max_length=50)] = None,
    stars_min: Annotated[int | None, Query(ge=1, le=5)] = None,
    page: Annotated[int, Query(ge=1)] = 1,
    page_size: Annotated[int, Query(ge=1, le=24)] = 12,
) -> VenueListResponse:
    return await venue_service.website_list(
        session,
        q=q,
        city=city,
        stars_min=stars_min,
        page=page,
        page_size=page_size,
    )


@router.get("/{city}/{slug}", response_model=VenueDetail)
async def venue_detail(
    session: Annotated[AsyncSession, Depends(get_db_session)],
    city: str,
    slug: str,
) -> VenueDetail:
    try:
        return await venue_service.website_detail(session, city=city, slug=slug)
    except VenueNotFoundError:
        return error_response(
            status_code=status.HTTP_404_NOT_FOUND,
            code="not_found",
            message="Venue not found.",
            details={"resource": "venue"},
        )
```
